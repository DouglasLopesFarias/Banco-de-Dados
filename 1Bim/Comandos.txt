//== exemplo para criar tabela
create table cidade(
    id int primary key,
	nome varchar(100),
	uf char(2)
);

//== TABELA DE VENDA_ITENS
CREATE TABLE VENDA_ITENS(
    PRODUTO_ID INT NOT NULL,
    VENDA_ID INT NOT NULL,
    VL_UNIT DECIMAL(10,2),
    QT INT NOT NULL,
    VL_TOTAL DECIMAL(10,2),
    PRIMARY KEY (PRODUTO_ID, VENDA_ID)
);

//== add chave estrangeira
ALTER TABLE VENDA_ITENS ADD CONSTRAINT  FK_VENDA_ITENS_VENDA FOREIGN KEY (VENDA_ID) REFERENCES VENDA(ID);
ALTER TABLE VENDA_ITENS ADD CONSTRAINT  FK_VENDA_ITENS_PRODUTO FOREIGN KEY(PRODUTO_ID) REFERENCES PRODUTO(ID);
