//== exemplo para criar tabela
CREATE TABLE CIDADE(
    ID INT PRIMARY KEY,
	NOME VARCHAR(100),
	UF CHAR(2)
);

CREATE TABLE FORNECEDOR(
	ID INT NOT NULL PRIMARY KEY,
	CIDADE_ID INT NOT NULL,
	NOME VARCHAR(100),
	ENDERECO VARCHAR(150),
	FONE VARCHAR(20),
	DATA_CAD DATE,	
	CONSTRAINT FK_FORNECEDOR_CIDADE FOREIGN  KEY(CIDADE_ID) REFERENCES CIDADE(ID)
);


CREATE TABLE COMPRA(
	ID INT NOT NULL PRiMARY KEY,
	FORNECEDOR_ID INT NOT NULL,
	DATA_EMICAO DATE,
	VALOR_TOTAL DECIMAL(10,2),
	FOREIGN  KEY(FORNECEDOR_ID) REFERENCES 	FORNECEDOR(ID)    
);

CREATE TABLE CLIENTE(
	ID INT NOT NULL PRIMARY KEY,
	CIDADE_ID INT NOT NULL,
	NOME VARCHAR(100),
	ENDERECO VARCHAR(150),
	FONE VARCHAR(20),
	DATA_NASC DATE,
	FOREIGN KEY(CIDADE_ID) REFERENCES CIDADE(ID)
);

CREATE TABLE PRODUTO(
  ID INT NOT NULL,
  DESCRICAO VARCHAR(150),
  VALOR DECIMAL(10,2),
  QT DECIMAL(10,3),
  PRIMARY KEY (ID)
);

CREATE TABLE VENDA(
	ID INT NOT NULL PRIMARY KEY,
	CLIENTE_ID INT NOT NULL,
	DATA_EMICAO DATE,
	VALOR_TOTAL DECIMAL(10,2),
	FOREIGN KEY(CLIENTE_ID) REFERENCES CLIENTE(id)	
);

CREATE TABLE VENDA_ITENS(
    PRODUTO_ID INT NOT NULL,
    VENDA_ID INT NOT NULL,
    VL_UNIT DECIMAL(10,2),
    QT INT NOT NULL,
    VL_TOTAL DECIMAL(10,2),
    PRIMARY KEY (PRODUTO_ID, VENDA_ID)
);

CREATE TABLE COMPRA_ITENS(
    PRODUTO_ID INT NOT NULL,
    COMPRA_ID INT NOT NULL,
    VL_UNIT DECIMAL(10,2),
    QT INT NOT NULL,
    VL_TOTAL DECIMAL(10,2),
    PRIMARY KEY (PRODUTO_ID, COMPRA_ID)
);
ALTER TABLE VENDA_ITENS ADD CONSTRAINT  FK_VENDA_ITENS_VENDA FOREIGN KEY (VENDA_ID) REFERENCES VENDA(ID);
ALTER TABLE VENDA_ITENS ADD CONSTRAINT  FK_VENDA_ITENS_PRODUTO FOREIGN KEY(PRODUTO_ID) REFERENCES PRODUTO(ID);
ALTER TABLE COMPRA_ITENS ADD CONSTRAINT  FK_COMPRA_ITENS_COMPRA FOREIGN KEY (COMPRA_ID) REFERENCES COMPRA(ID);
ALTER TABLE COMPRA_ITENS ADD CONSTRAINT  FK_COMPRA_ITENS_PRODUTO FOREIGN KEY(PRODUTO_ID) REFERENCES PRODUTO(ID);
